return function(Window,WindUI) local Players=game:GetService("Players") local LocalPlayer=Players.LocalPlayer local RunService=game:GetService("RunService") local HttpService=game:GetService("HttpService") local SAVE_FILE="windui_teleport_coords.json" local savedCoords={} local selectedIndex=nil local selectedName=nil local tab=Window:Tab({Title="传送",Icon="map-pin"}) local teleportSection=tab:Section({Title="坐标传送",Box=true}) local inputValue="" local loopConnection=nil teleportSection:Input({Title="输入坐标",Placeholder="例如: 100, 200, 300",Callback=function(text) inputValue=text end}) teleportSection:Button({Title="传送",Callback=function() if inputValue=="" then WindUI:Notify({Title="提示",Content="请先输入坐标",Duration=2,Icon="alert"}) return end local parts={} for part in inputValue:gmatch("[^,，%s]+") do local num=tonumber(part) if num then table.insert(parts,num) end end if #parts>=3 then local char=LocalPlayer.Character if char and char:FindFirstChild("HumanoidRootPart") then char.HumanoidRootPart.CFrame=CFrame.new(Vector3.new(parts[1],parts[2],parts[3])) WindUI:Notify({Title="传送成功",Content=string.format("已传送到 (%.2f, %.2f, %.2f)",parts[1],parts[2],parts[3]),Duration=2,Icon="check"}) end else WindUI:Notify({Title="格式错误",Content="请输入如: 100,200,300",Duration=2,Icon="alert"}) end end}) teleportSection:Divider() local coordDisplay=teleportSection:Paragraph({Title="当前坐标",Desc="X: 0.00  Y: 0.00  Z: 0.00"}) local function updateCoordDisplay() local char=LocalPlayer.Character if char and char:FindFirstChild("HumanoidRootPart") then local pos=char.HumanoidRootPart.Position coordDisplay:SetDesc(string.format("X: %.2f  Y: %.2f  Z: %.2f",pos.X,pos.Y,pos.Z)) end end RunService.RenderStepped:Connect(updateCoordDisplay) teleportSection:Button({Title="复制当前坐标",Callback=function() local char=LocalPlayer.Character if char and char:FindFirstChild("HumanoidRootPart") then local pos=char.HumanoidRootPart.Position local coordText=string.format("%.2f,%.2f,%.2f",pos.X,pos.Y,pos.Z) if setclipboard then setclipboard(coordText) WindUI:Notify({Title="复制成功",Content=coordText,Duration=2,Icon="check"}) else WindUI:Notify({Title="复制失败",Content="当前环境不支持复制",Duration=2,Icon="x"}) end else WindUI:Notify({Title="错误",Content="未找到角色",Duration=2,Icon="alert"}) end end}) teleportSection:Toggle({Title="循环传送",Callback=function(val) if val then if loopConnection then return end loopConnection=RunService.Heartbeat:Connect(function() if inputValue=="" then return end local parts={} for part in inputValue:gmatch("[^,，%s]+") do local num=tonumber(part) if num then table.insert(parts,num) end end if #parts>=3 then local char=LocalPlayer.Character if char and char:FindFirstChild("HumanoidRootPart") then char.HumanoidRootPart.CFrame=CFrame.new(Vector3.new(parts[1],parts[2],parts[3])) end end end) else if loopConnection then loopConnection:Disconnect() loopConnection=nil end end end}) local coordSection=tab:Section({Title="坐标收藏",Box=true}) local saveNameValue="" local renameValue="" local function saveCoords() if writefile then writefile(SAVE_FILE,HttpService:JSONEncode(savedCoords)) end end local function loadCoords() if readfile and isfile and isfile(SAVE_FILE) then local ok,data=pcall(function() return HttpService:JSONDecode(readfile(SAVE_FILE)) end) if ok and type(data)=="table" then savedCoords=data end end end local function findIndexByName(name) for i,c in ipairs(savedCoords) do if c.name==name then return i end end return nil end local function nameExists(name) return findIndexByName(name)~=nil end local function generateAutoName() local i=1 while true do local n="坐标"..i if not nameExists(n) then return n end i=i+1 end end local function getDropdownValues() local vals={} for _,c in ipairs(savedCoords) do table.insert(vals,c.name) end return vals end local coordDropdown=nil local function refreshDropdown() if coordDropdown then coordDropdown:Refresh(getDropdownValues()) end end loadCoords() local hasPreset=false for _,c in ipairs(savedCoords) do if c._preset and c.name=="传送虚空" then hasPreset=true break end end if not hasPreset then table.insert(savedCoords,{name="传送虚空",x=0,y=100000000,z=0,_preset=true}) saveCoords() end coordSection:Input({Title="收藏名称（可选）",Placeholder="留空自动命名为 坐标1、坐标2…",Callback=function(text) saveNameValue=text end}) coordSection:Button({Title="保存当前坐标",Callback=function() local char=LocalPlayer.Character if not (char and char:FindFirstChild("HumanoidRootPart")) then WindUI:Notify({Title="错误",Content="未找到角色",Duration=2,Icon="alert"}) return end local pos=char.HumanoidRootPart.Position local name=saveNameValue if name=="" then name=generateAutoName() elseif nameExists(name) then WindUI:Notify({Title="名称重复",Content="已存在同名坐标",Duration=2,Icon="alert"}) return end table.insert(savedCoords,{name=name,x=tonumber(string.format("%.2f",pos.X)),y=tonumber(string.format("%.2f",pos.Y)),z=tonumber(string.format("%.2f",pos.Z))}) saveCoords() refreshDropdown() WindUI:Notify({Title="已收藏",Content=name,Duration=2,Icon="check"}) end}) coordDropdown=coordSection:Dropdown({Title="已保存坐标",Values=getDropdownValues(),Callback=function(opt) if not opt then char:FindFirstChild("HumanoidRootPart") then char.HumanoidRootPart.CFrame=CFrame.new(Vector3.new(c.x,c.y,c.z)) WindUI:Notify({Title="传送成功",Content=c.name,Duration=2,Icon="check"}) end end}) local selectedLoopConn=nil coordSection:Toggle({Title="循环传送到选中坐标",Callback=function(val) if val then if selectedLoopConn then return end selectedLoopConn=RunService.Heartbeat:Connect(function() if not selectedIndex or not savedCoords[selectedIndex] then return end local c=savedCoords[selectedIndex] local char=LocalPlayer.Character if char and char:FindFirstChild("HumanoidRootPart") then char.HumanoidRootPart.CFrame=CFrame.new(Vector3.new(c.x,c.y,c.z)) end end) else if selectedLoopConn then selectedLoopConn:Disconnect() selectedLoopConn=nil end end end}) coordSection:Input({Title="新名称（重命名用）",Placeholder="输入新名称",Callback=function(text) renameValue=text end}) coordSection:Button({Title="重命名选中坐标",Callback=function() if not selectedIndex or not savedCoords[selectedIndex] then WindUI:Notify({Title="错误",Content="请先在下拉菜单选择坐标",Duration=2,Icon="alert"}) return end if renameValue=="" then WindUI:Notify({Title="提示",Content="请先输入新名称",Duration=2,Icon="alert"}) return end if nameExists(renameValue) and renameValue~=savedCoords[selectedIndex].name then WindUI:Notify({Title="名称重复",Content="已存在同名坐标",Duration=2,Icon="alert"}) return end savedCoords[selectedIndex].name=renameValue saveCoords() selectedIndex=nil selectedName=nil refreshDropdown() WindUI:Notify({Title="已重命名",Content=renameValue,Duration=2,Icon="check"}) end}) coordSection:Button({Title="删除选中坐标",Callback=function() if not selectedIndex or not savedCoords[selectedIndex] then WindUI:Notify({Title="错误",Content="请先在下拉菜单选择坐标",Duration=2,Icon="alert"}) return end local c=savedCoords[selectedIndex] if c._preset then WindUI:Notify({Title="禁止",Content="预设坐标不可删除",Duration=2,Icon="x"}) return end local idx=selectedIndex Window:Dialog({Title="确认删除",Content="确定要删除坐标「"..c.name.."」吗？此操作不可恢复。",Buttons={{Title="取消",Callback=function() end},{Title="确认删除",Callback=function() table.remove(savedCoords,idx) selectedIndex=nil selectedName=nil if selectedLoopConn then selectedLoopConn:Disconnect() selectedLoopConn=nil end saveCoords() refreshDropdown() WindUI:Notify({Title="已删除",Content=c.name,Duration=2,Icon="check"}) end}}}) end}) return tab end
